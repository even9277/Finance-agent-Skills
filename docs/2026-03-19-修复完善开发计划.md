## P1：报告模式股票名称/代码提取增强

### 1.1 现状分析

**问题根因：**

当前 `backend/services/agent_service.py` 的 `extract_stock_info()` 只依赖硬编码的正则表达式来提取股票名称和代码：

```python
# agent_service.py L116-161 — 当前实现
def extract_stock_info(query: str) -> tuple[str | None, str | None]:
    patterns_with_both = [
        r'请帮我分析一下\s*([^（(]+?)\s*[（(](\d{5,6})[)）]',
        r'分析一下\s*([^（(]+?)\s*[（(](\d{5,6})[)）]',
        # ... 仅支持"分析"类句式
    ]
```

**缺陷：**

1. **只支持有限的句式模板** — "帮我看看比亚迪最近怎么样" 这类口语化表达无法匹配
2. **只有股票名称没有代码时，无法反查代码** — 用户说"分析贵州茅台"，正则能提取到名称，但 `stock_code` 为 `None`，各 Agent 只能用 `Unknown` 作为股票代码调用 MCP 工具，**导致 MCP 工具报错或返回无数据**
3. **不支持简称/别名** — 如"茅台"→"贵州茅台"→"sh.600519"
4. **`agent_service.py` 和 `main.py` 各自维护一套正则**，逻辑不同步

### 1.2 修复方案

#### 方案总体思路

引入一个 **`stock_resolver` 预处理模块**，在工作流启动前完成"名称 → 代码"的确定性解析：

```
用户输入 → [正则提取(增强版)] → [BaoStock 名称反查(新增)] → [LLM 兜底提取(新增)] → 确定的 (company_name, stock_code)
```

三层递进策略：
- **第一层**：增强正则（快、零成本）
- **第二层**：BaoStock `get_all_stock()` 反查（准确、耗时约1-2s）
- **第三层**：LLM 抽取兜底（处理口语化/模糊输入）

#### 1.2.1 新建 `stock_resolver.py`

**文件位置**：`backend/services/stock_resolver.py`

```python
"""
stock_resolver.py — 股票名称/代码预解析模块

职责：在多 Agent 工作流启动前，将用户自然语言中的股票标识
解析为确定的 (company_name, stock_code) 二元组。

三层递进策略：
  L1 - 增强正则：覆盖"分析/看看/了解/怎么样"等常见句式 + 纯代码
  L2 - BaoStock 名称反查：用户只给名称时，查 get_all_stock 匹配全称/简称
  L3 - LLM 兜底：正则和反查都失败时，调用 LLM 做一次结构化抽取

设计原则：
  - 本模块是纯函数/工具模块，不修改任何 Agent 或工作流逻辑
  - 错误不阻断流程：任何一层失败均静默降级到下一层
  - 可被 agent_service.py 和 main.py 共同调用，消除重复维护
"""

import os
import re
import logging
from typing import Optional
from functools import lru_cache

logger = logging.getLogger("stock_resolver")


# ═══════════════════════════════════════════════════════════════
# L1: 增强正则提取
# ═══════════════════════════════════════════════════════════════

# 合并 agent_service.py + main.py 的所有模式，并新增口语化模式
_PATTERNS_BOTH = [
    # 名称(代码) 格式
    r'([^（(,，\s]{2,8}?)\s*[（(](\d{6})[)）]',
    # 代码.名称 格式：600519.贵州茅台
    r'(\d{6})[.\s]+([^\s,，]{2,8})',
]

_PATTERNS_CODE_ONLY = [
    r'\b(\d{6})\b',          # 纯6位数字
    r'[shSH]{2}\.?(\d{6})',  # sh.600519 / SH600519
    r'[szSZ]{2}\.?(\d{6})',  # sz.000001
]

_PATTERNS_NAME_ONLY = [
    r'(?:分析|看看|了解|研究|查一下|查查|评估|调研|帮我看)\s*(?:一下\s*)?([^\d（）()\s,，]{2,8}?)(?:\s*(?:的|这只|这个|股票|怎么样|最近|走势|基本面|技术面|财务)|\s*$)',
    r'([^\d（）()\s,，]{2,8}?)\s*(?:这只|这个|的)?\s*(?:股票|走势|行情|基本面)',
    r'(?:关于|对于)\s*([^\d（）()\s,，]{2,8}?)\s*(?:的|，)',
]

_STOP_WORDS = frozenset([
    '的', '这个', '这只', '一下', '看看', '了解', '分析', '帮我', '我想',
    '给我', '财务状况', '投资价值', '基本面', '技术面', '请', '能否',
    '最近', '怎么样', '研究', '调研', '评估', '查一下', '查查',
])


def _regex_extract(query: str) -> tuple[Optional[str], Optional[str]]:
    """L1: 纯正则提取，返回 (company_name, stock_code)"""
    company_name = None
    stock_code = None

    # 尝试同时提取名称和代码
    for pattern in _PATTERNS_BOTH:
        m = re.search(pattern, query)
        if m:
            g1, g2 = m.group(1).strip(), m.group(2).strip()
            if g1.isdigit():
                stock_code, company_name = g1, g2
            else:
                company_name, stock_code = g1, g2
            break

    # 仅代码
    if not stock_code:
        for pattern in _PATTERNS_CODE_ONLY:
            m = re.search(pattern, query)
            if m:
                stock_code = m.group(1)
                break

    # 仅名称
    if not company_name:
        for pattern in _PATTERNS_NAME_ONLY:
            m = re.search(pattern, query)
            if m:
                candidate = m.group(1).strip()
                for w in _STOP_WORDS:
                    candidate = candidate.replace(w, '').strip()
                if len(candidate) >= 2:
                    company_name = candidate
                    break

    return company_name, stock_code


# ═══════════════════════════════════════════════════════════════
# L2: BaoStock 名称 → 代码反查
# ═══════════════════════════════════════════════════════════════

_STOCK_NAME_MAP: dict[str, str] = {}  # name -> "sh.600519"
_STOCK_NAME_MAP_READY = False


def _init_stock_name_map() -> None:
    """
    一次性加载 BaoStock 全市场股票列表，构建 名称→代码 映射表。
    在首次调用时触发，后续走内存缓存。
    
    失败时静默降级（映射表为空，L2 跳过）。
    """
    global _STOCK_NAME_MAP, _STOCK_NAME_MAP_READY
    if _STOCK_NAME_MAP_READY:
        return
    _STOCK_NAME_MAP_READY = True
    
    try:
        import baostock as bs
        from src.utils import baostock_login_context
        
        with baostock_login_context():
            rs = bs.query_all_stock()
            while rs.next():
                row = rs.get_row_data()
                # row: [code, tradeStatus, code_name]
                if len(row) >= 3:
                    code, name = row[0], row[2]
                    if name and code:
                        _STOCK_NAME_MAP[name] = code
                        # 简称映射：去掉常见后缀
                        short = name.replace('股份', '').replace('集团', '').replace('控股', '').strip()
                        if short and short != name and len(short) >= 2:
                            _STOCK_NAME_MAP.setdefault(short, code)
        
        logger.info(f"[stock_resolver] BaoStock 名称映射加载完成: {len(_STOCK_NAME_MAP)} 条")
    except Exception as exc:
        logger.warning(f"[stock_resolver] BaoStock 名称映射加载失败（L2 降级）: {exc}")


def _baostock_lookup(company_name: str) -> Optional[str]:
    """
    L2: 通过名称在全市场列表中查找股票代码。
    支持精确匹配 + 包含匹配（优先精确）。
    返回 BaoStock 格式的代码，如 "sh.600519"。
    """
    _init_stock_name_map()
    if not _STOCK_NAME_MAP:
        return None
    
    # 精确匹配
    if company_name in _STOCK_NAME_MAP:
        return _STOCK_NAME_MAP[company_name]
    
    # 包含匹配（名称是某个全称的子串）
    candidates = []
    for name, code in _STOCK_NAME_MAP.items():
        if company_name in name or name in company_name:
            candidates.append((name, code))
    
    if len(candidates) == 1:
        return candidates[0][1]
    elif len(candidates) > 1:
        # 多个匹配时取最短名称（最精确）
        candidates.sort(key=lambda x: len(x[0]))
        logger.info(
            f"[stock_resolver] L2 多个匹配: query='{company_name}', "
            f"选择='{candidates[0][0]}' ({candidates[0][1]})"
        )
        return candidates[0][1]
    
    return None


# ═══════════════════════════════════════════════════════════════
# L3: LLM 兜底抽取
# ═══════════════════════════════════════════════════════════════

_LLM_EXTRACT_PROMPT = """你是一个 A 股股票信息提取助手。从用户的自然语言输入中提取股票信息。

用户输入：{query}

请从中提取：
1. company_name：公司全称或简称（如"贵州茅台"、"比亚迪"）
2. stock_code：6位数字股票代码（如"600519"、"002594"），如果用户没有给出代码则填 null

严格输出 JSON 格式，不要输出任何其他内容：
{{"company_name": "xxx", "stock_code": "123456"}}

如果无法识别任何股票信息，输出：
{{"company_name": null, "stock_code": null}}"""


async def _llm_extract(query: str) -> tuple[Optional[str], Optional[str]]:
    """
    L3: 调用 LLM 做结构化抽取（仅在 L1+L2 均失败时使用）。
    使用与各 Agent 相同的 OpenAI 兼容 API。
    """
    import json as _json
    try:
        from langchain_openai import ChatOpenAI
        from langchain_core.messages import HumanMessage
        
        api_key = os.getenv("OPENAI_COMPATIBLE_API_KEY")
        base_url = os.getenv("OPENAI_COMPATIBLE_BASE_URL")
        model_name = os.getenv("OPENAI_COMPATIBLE_MODEL")
        
        if not all([api_key, base_url, model_name]):
            return None, None
        
        llm = ChatOpenAI(
            model=model_name, api_key=api_key, base_url=base_url,
            temperature=0, max_tokens=200,
        )
        
        resp = await llm.ainvoke([HumanMessage(content=_LLM_EXTRACT_PROMPT.format(query=query))])
        text = resp.content.strip()
        
        # 提取 JSON（兼容 LLM 可能输出 markdown code block）
        if '```' in text:
            text = text.split('```')[1]
            if text.startswith('json'):
                text = text[4:]
        
        data = _json.loads(text)
        return data.get("company_name"), data.get("stock_code")
    except Exception as exc:
        logger.warning(f"[stock_resolver] LLM 抽取失败（L3 降级）: {exc}")
        return None, None


# ═══════════════════════════════════════════════════════════════
# 统一入口
# ═══════════════════════════════════════════════════════════════

def add_exchange_prefix(stock_code: str) -> str:
    """为纯数字股票代码添加交易所前缀。"""
    code = stock_code.replace('sh.', '').replace('sz.', '')
    if code.startswith('6'):
        return f"sh.{code}"
    elif code.startswith('0') or code.startswith('3'):
        return f"sz.{code}"
    return stock_code


async def resolve_stock(query: str) -> tuple[Optional[str], Optional[str]]:
    """
    统一入口：从用户自然语言中解析股票信息。
    
    返回：(company_name, stock_code)
    - stock_code 已包含交易所前缀（如 "sh.600519"）
    - 任一字段可能为 None（未能解析时）
    
    三层递进：L1 正则 → L2 BaoStock 反查 → L3 LLM 兜底
    """
    # L1: 正则
    company_name, stock_code = _regex_extract(query)
    
    # 如果有代码但无名称 → 补名称（BaoStock 反查）
    if stock_code and not company_name:
        prefixed = add_exchange_prefix(stock_code)
        _init_stock_name_map()
        for name, code in _STOCK_NAME_MAP.items():
            if code == prefixed:
                company_name = name
                break
    
    # 如果有名称但无代码 → L2 BaoStock 名称反查
    if company_name and not stock_code:
        resolved_code = _baostock_lookup(company_name)
        if resolved_code:
            stock_code = resolved_code
            logger.info(f"[stock_resolver] L2 反查成功: '{company_name}' → {stock_code}")
    
    # 如果 L1+L2 都未能解析出代码 → L3 LLM 兜底
    if not stock_code:
        llm_name, llm_code = await _llm_extract(query)
        if llm_code:
            stock_code = llm_code
            logger.info(f"[stock_resolver] L3 LLM 抽取: code={llm_code}")
        if llm_name and not company_name:
            company_name = llm_name
        
        # LLM 给出名称后再尝试一次 L2 反查
        if company_name and not stock_code:
            resolved_code = _baostock_lookup(company_name)
            if resolved_code:
                stock_code = resolved_code
    
    # 统一添加交易所前缀
    if stock_code and not stock_code.startswith(('sh.', 'sz.')):
        stock_code = add_exchange_prefix(stock_code)
    
    logger.info(f"[stock_resolver] 最终结果: company='{company_name}', code='{stock_code}'")
    return company_name, stock_code
```

## P2：报告模式 LTM 注入修复

### 2.1 现状分析

**问题根因（共 2 处致命缺陷）：**

**缺陷 A**：`agent_service.py` 的 `_get_workflow()` **完全没有集成 LTM 节点**。

```python
# agent_service.py L61-110 — 当前工作流编译
def _get_workflow():
    # 只检查了 enable_stm，完全没有 enable_memory 的逻辑！
    enable_stm = settings.enable_stm
    # ...
    if enable_stm:
        # 只插入 STM 节点
        wf.add_node("prepare_summary_context", prepare_summary_context)
        wf.add_node("maybe_summarize_state", maybe_summarize_state)
        # ...
    else:
        # 原始模式
        # ...
```

对比 `main.py`（CLI 模式），后者正确处理了 4 种组合（无 STM 无 LTM / 仅 STM / 仅 LTM / 都启用）。但 `agent_service.py` 只处理了 2 种（无 STM / 仅 STM）。

**缺陷 B**：`_build_initial_state()` **未将 `user_id` 设为 `memory_user_id`**。

```python
# agent_service.py L186 — 当前实现
return AgentState(messages=[], data=data, metadata={}, **make_stm_defaults())
# make_stm_defaults() 返回 memory_user_id="" → memory_read_node 因 user_id 为空而跳过
```

而 `run_report_task` 明明接收了 `user_id` 参数（L196），却没有传递给初始状态。

### 2.2 修复方案

#### 2.2.1 `_get_workflow()` 加入 LTM 节点

## P3：对话历史摘要 → 长期记忆自动更新逻辑优化

### 3.1 现状分析

**问题根因：**

当前 `maybe_update_ltm_from_chat()` 的入队逻辑和 `ltm_worker` 的处理逻辑存在以下问题：

#### 3.1.1 入队阶段（`chat_service.py`）

```python
# chat_service.py L1065-1069 — 当前消息构建
messages_for_ltm = [
    {"role": m.role, "content": m.content[:800]}
    for m in pending_msgs
]
```

**问题**：
1. **直接将原始对话消息扔给 Mem0** — 对话中大量是闲聊、确认、追问，有效信息密度极低
2. **没有利用 running_summary** — 虽然代码在 L1077-1081 将摘要 prepend 到消息列表，但这只是作为一条 `system` 消息添加，Mem0 的 `infer=False` 模式下这条摘要不会被语义理解
3. **触发时机粗糙** — 满 5 条 user 消息或 `turn_count % 10 == 0` 就触发，不考虑对话内容是否真的包含画像相关信息

#### 3.1.2 处理阶段（`ltm_worker.py`）

```python
# ltm_worker.py — _mem0_add_safe 中 infer=False
mem0_client.add(messages=payload_messages, user_id=user_id, metadata=metadata, infer=False)
```

**问题**：
1. **`infer=False` 禁用了 Mem0 的 LLM 事实抽取** — 这是之前为了避免 `dict.replace` 崩溃的权宜之计，但导致 Mem0 无法从对话中提取结构化信息
2. **入队的原始消息质量不高** — 即使 `infer=True`，原始闲聊也难以提取有价值的投资画像

### 3.2 修复方案

#### 总体思路

将"先入队原始消息，worker 端靠 Mem0 抽取"的模式，改为：
**"入队前先用 LLM 做画像要素抽取，只将结构化抽取结果入队"**

```
对话消息 → [LLM 画像要素抽取(新增)] → 结构化 facts → enqueue → ltm_worker → Mem0
```

#### 3.2.1 新建 `profile_extractor.py` — 画像要素抽取器

**文件位置**：`backend/services/profile_extractor.py`

```python
"""
profile_extractor.py — 对话画像要素抽取器

职责：从一段对话消息中提取用户投资画像的结构化更新信号。
仅在对话中确实包含画像相关信息时才返回非空结果。

设计原则：
  - 抽取结果必须映射到 user_invest_profiles 的标准字段/枚举
  - 低置信度的推断不输出（宁缺勿滥）
  - 本模块不直接写 DB，只返回结构化结果供调用方决策
"""

import os
import json
import logging
from typing import Optional

logger = logging.getLogger("profile_extractor")

_EXTRACT_PROMPT = """你是一个投资用户画像分析师。请分析以下对话片段，提取用户的投资偏好信息。

## 可提取的字段和允许值

1. risk_level: 风险偏好
   允许值: conservative(保守), moderate(稳健), balanced(平衡), aggressive(进取), speculative(激进)

2. investment_horizon: 投资周期
   允许值: ultra_short(超短线), short(短线), swing(波段), long(中长线)

3. sectors: 关注板块（数组）
   允许值: 科技/半导体, AI/大模型, 新能源/光伏, 医药/生物, 消费/白酒, 金融/银行, 
           军工/航天, 房地产, 汽车/新能源车, 黄金/贵金属, 红利/高股息, 周期/资源

4. expected_return_min / expected_return_max: 期望收益率范围（数字，百分比）

5. constraints: 约束条件（数组，如 "不碰ST股"）

6. response_pref: 回答偏好
   允许值: concise(简洁), detailed(详细), balanced(平衡), risk_first(先讲风险)

## 对话内容

{conversation}

## 要求

1. **只提取用户明确表达或强烈暗示的偏好**，不要过度推断
2. 如果用户只是询问某只股票，不代表他关注该板块（除非有其他上下文支持）
3. 如果对话中完全没有画像相关信息，返回空对象
4. 置信度标注：对每个提取的字段给出 high/medium 置信度

严格输出 JSON，不要任何其他内容：
{{
  "updates": [
    {{"field": "risk_level", "value": "conservative", "confidence": "high", "evidence": "用户说'我比较保守'"}},
    ...
  ],
  "has_profile_signal": true/false
}}

如果对话中无任何投资偏好信号，输出：
{{"updates": [], "has_profile_signal": false}}"""


async def extract_profile_updates(
    messages: list[dict],
    running_summary: str = "",
) -> dict:
    """
    从对话消息中抽取画像更新信号。
    
    Args:
        messages: [{"role": "user"/"assistant", "content": "..."}]
        running_summary: 当前会话的 STM 摘要（可选上下文）
    
    Returns:
        {
            "updates": [{"field": "risk_level", "value": "...", "confidence": "high", "evidence": "..."}],
            "has_profile_signal": bool
        }
    """
    try:
        from langchain_openai import ChatOpenAI
        from langchain_core.messages import HumanMessage

        api_key = os.getenv("OPENAI_COMPATIBLE_API_KEY")
        base_url = os.getenv("OPENAI_COMPATIBLE_BASE_URL")
        model_name = os.getenv("OPENAI_COMPATIBLE_MODEL")

        if not all([api_key, base_url, model_name]):
            return {"updates": [], "has_profile_signal": False}

        # 构建对话文本
        conv_parts = []
        if running_summary:
            conv_parts.append(f"[对话摘要] {running_summary[:500]}")
        for m in messages:
            role_label = "用户" if m["role"] == "user" else "助手"
            conv_parts.append(f"{role_label}: {m['content'][:600]}")
        conversation_text = "\n".join(conv_parts)

        llm = ChatOpenAI(
            model=model_name, api_key=api_key, base_url=base_url,
            temperature=0, max_tokens=500,
        )

        resp = await llm.ainvoke([
            HumanMessage(content=_EXTRACT_PROMPT.format(conversation=conversation_text))
        ])
        text = resp.content.strip()

        # 解析 JSON
        if '```' in text:
            text = text.split('```')[1]
            if text.startswith('json'):
                text = text[4:]
        
        result = json.loads(text)
        
        # 过滤：只保留 high/medium 置信度的更新
        if result.get("updates"):
            result["updates"] = [
                u for u in result["updates"]
                if u.get("confidence") in ("high", "medium")
            ]
            result["has_profile_signal"] = len(result["updates"]) > 0
        
        return result
    except Exception as exc:
        logger.warning(f"[profile_extractor] 抽取失败: {exc}")
        return {"updates": [], "has_profile_signal": False}
```

```python
# chat_service.py — 重构 maybe_update_ltm_from_chat

async def maybe_update_ltm_from_chat(

            user_msgs_count = sum(1 for m in pending_msgs if m.role == "user")

            trigger_by_count = user_msgs_count >= _LTM_TRIGGER_MSG_COUNT
            trigger_by_compress = (turn_count > 0 and turn_count % 10 == 0)

            if not trigger_by_count and not trigger_by_compress:
                return

            # 最小间隔检查（保持原有逻辑）
            # ... (同现有代码) ...

            # ═══ 新增：画像要素抽取 ═══
            messages_for_extract = [
                {"role": m.role, "content": m.content[:800]}
                for m in pending_msgs

                messages=messages_for_extract,
                running_summary=running_summary,

                for update in extraction["updates"]:

                    value = update.get("value")
                    if field and value:

                                user_id=user_id,

                            )
                        except Exception as uf_exc:

            # ═══ 语义增强入队（Mem0） ═══
            # 只在有画像信号时入队，避免无意义的入队
            if extraction.get("has_profile_signal"):
                messages_for_ltm = messages_for_extract.copy()
                if running_summary:
                    messages_for_ltm.insert(0, {
                        "role": "system",
                        "content": f"[对话摘要] {running_summary[:500]}",
                    })

                    "evidence_ref": ",".join([str(m.id) for m in pending_msgs]),
                    "active": True,
                    "updated_by": "llm",
                    "confidence": 0.7,
                    "extracted_fields": [u["field"] for u in extraction.get("updates", [])],
                }

                await MemoryService.enqueue_add_conversation(
                    user_id=user_id,
                    messages=messages_for_ltm,

                )
            else:
                logger.debug(
                    f"[LTM-chat] 对话无画像信号，跳过 Mem0 入队: "
                    f"session={session_id[:8]}..."
                )

            # 标记已处理
            for msg in pending_msgs:
                msg.used_for_ltm = True
            await db.commit()

    except Exception as exc:
        logger.error(f"[LTM-chat] maybe_update_ltm_from_chat 异常: {exc}", exc_info=True)

#### 3.2.3 `ltm_worker.py` — 恢复 `infer=True` 并加强保护

之前因为 `infer=True` 导致 `dict.replace` 崩溃而关闭了推理，现在入队的消息质量已经提升，可以有选择地恢复：

```python
# ltm_worker.py — _mem0_add_safe 修改

async def _mem0_add_safe(mem0_client, *, user_id, messages, metadata):
    """
    修改策略：
    - 如果 metadata 中有 extracted_fields（说明已经过 profile_extractor 预抽取），
      使用 infer=False（避免重复抽取 + 崩溃风险）
    - 否则使用 infer=True 让 Mem0 自行抽取
    """
    has_pre_extraction = bool(metadata.get("extracted_fields"))
    infer = not has_pre_extraction

    dict_messages, text_messages = _coerce_messages_for_mem0(messages)
    single_text = "\n".join(text_messages) if text_messages else ""

    async def _call(payload_messages):
        await asyncio.wait_for(
            mem0_client.add(
                messages=payload_messages,
                user_id=user_id,
                metadata=metadata,
                infer=infer,
            ),
            timeout=LTM_WORKER_TASK_TIMEOUT_SEC,
        )

    # 三层回退保持不变
    try:
        await _call(dict_messages)
        return
    except Exception as exc1:
        if isinstance(exc1, asyncio.TimeoutError):
            raise
        # 如果 infer=True 导致崩溃，自动回退到 infer=False
        if infer and ("replace" in str(exc1) or "string indices" in str(exc1)):
            logger.warning(f"[ltm_worker] infer=True 崩溃，回退到 infer=False: {exc1}")
            infer = False
            try:
                await asyncio.wait_for(
                    mem0_client.add(
                        messages=text_messages or single_text,
                        user_id=user_id,
                        metadata=metadata,
                        infer=False,
                    ),
                    timeout=LTM_WORKER_TASK_TIMEOUT_SEC,
                )
                return
            except Exception as exc2:
                raise exc2
        raise exc1
```
